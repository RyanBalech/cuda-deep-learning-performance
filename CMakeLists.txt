cmake_minimum_required(VERSION 3.24)
project(cuda_dl_performance LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_RUNTIME_LIBRARY None)
include(CTest)
find_package(CUDAToolkit REQUIRED)
add_library(cpu_baselines cpp/cpu_baselines.cpp)
target_include_directories(cpu_baselines PUBLIC include)
add_library(cuda_kernels cuda/matmul.cu)
target_include_directories(cuda_kernels PUBLIC include)
target_link_libraries(cuda_kernels PUBLIC CUDA::cudart)
add_library(cublas_baseline cuda/cublas_baseline.cu)
target_include_directories(cublas_baseline PUBLIC include)
target_link_libraries(cublas_baseline PUBLIC CUDA::cublas CUDA::cudart)
add_executable(gemm_benchmark benchmarks/gemm_benchmark.cu)
target_link_libraries(gemm_benchmark PRIVATE cuda_kernels cublas_baseline cpu_baselines)
if(BUILD_TESTING)
  add_executable(gemm_correctness tests/gemm_correctness.cu)
  target_link_libraries(gemm_correctness PRIVATE cuda_kernels cublas_baseline cpu_baselines)
  add_test(NAME gemm_correctness COMMAND gemm_correctness)
endif()
